```python
# async python streaming + exponential backoff (v2 stream API)
import asyncio
from redwood import Client

async def simple_stream():
    client = Client()
    attempt = 0
    max_attempts = 3
    backoff = 0.25
    while True:
        try:
            async with client.stream("gpt-neo-1", prompt="hi") as s:
                async for piece in s:
                    print(piece.text, end="")
            break
        except Exception as e:
            attempt += 1
            if attempt >= max_attempts:
                raise
            await asyncio.sleep(backoff)
            backoff *= 2
```

elyse: TS snippet is in the PR, copied here:
```ts
// typescript streaming with jittered retry
import { Redwood } from "@redwood/inference"

async function run() {
  const client = new Redwood()
  let tries = 0
  while (tries < 4) {
    try {
      const stream = await client.stream("gpt-neo-1", { prompt: "hi" })
      for await (const chunk of stream) {
        process.stdout.write(chunk.text)
      }
      return
    } catch (err) {
      const backoff = (Math.random() + 0.5) * Math.pow(2, tries) * 100
      await new Promise(r => setTimeout(r, backoff))
      tries++
      if (tries === 4) throw err
    }
  }
}
```

```go
  resp, err := client.List(ctx, &ListParams{Cursor: cursor})
```
