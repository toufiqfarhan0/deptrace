```sql
INSERT INTO console_route_metrics_v2 (event_id, canonical_route_id, history_version, payload, ingested_at)
SELECT event_id, canonical_route_id, 2, payload, now()
FROM console_route_metrics_v1 v1
WHERE v1.event_ts >= $start AND v1.event_ts < $end

    SELECT 1 FROM console_route_metrics_v2 v2
    WHERE v2.event_id = v1.event_id AND v2.history_version = 2
  );
```
