```go
// simplified streaming + toolcall pseudo-example
client := redwood.NewClient("$REDWOOD_KEY")
stream := client.Stream(context.Background(), redwood.StreamRequest{
  Model: "gpt-stream-1",
  Prompt: "Update inventory for order #1234",
  Tools: []redwood.Tool{{Name: "inventory.update", Schema: "{\"id\":\"string\",\"qty\":\"int\"}"}},
})
for msg := range stream.Messages() {
  if msg.Type == redwood.MsgFunctionCall {
    // call inventory service
    call := msg.FunctionCall
    // respond with JSON payload
    stream.SendFunctionResponse(call.Name, `{\"id\":\"sku-123\",\"qty\":10}`)
  }
  if msg.Type == redwood.MsgOutput {
    fmt.Println(string(msg.Data)) // structured JSON output
  }
}
```
