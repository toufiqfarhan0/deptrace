```sql
-- untagged_gpu_spend_last_24h
SELECT project_id, sku_description, SUM(cost) AS cost
FROM `billing.gcp_line_items`
WHERE sku_description LIKE '%GPU%'
  AND DATE(usage_start_time) = DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY)
  AND NOT EXISTS (SELECT 1 FROM UNNEST(labels) l WHERE l.key IN ('team','env'))
GROUP BY project_id, sku_description
ORDER BY cost DESC
LIMIT 50;
```
