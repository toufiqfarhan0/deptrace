    gpg --sign --default-key $REPO_SIGN_KEY -o evidence-${GITHUB_RUN_ID}.tgz.sig evidence-${GITHUB_RUN_ID}.tgz && \
    gsutil cp evidence-${GITHUB_RUN_ID}.tgz* gs://internal-sdlc-evidence/$REPO/
```
