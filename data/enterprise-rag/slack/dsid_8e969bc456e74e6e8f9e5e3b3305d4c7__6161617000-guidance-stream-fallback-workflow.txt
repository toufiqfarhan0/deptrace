```ts
// TS SDK >=3.2
async function sendWithRetry(req) {
  for (let i=0;i<3;i++) {
    const res = await client.generate(req)
    if (res.status===429) {
      const delay = parseInt(res.headers['retry-after']||res.headers['x-rw-retry']||'1')
      await sleep(delay*1000 * Math.pow(2,i))
      continue
    }
    return res
  }
  throw new Error('rate limited')
}
```
