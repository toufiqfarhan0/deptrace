sam: We did something similar for GCP marketplace — module exposes a 'redwood_provider' map and uses locals to select provider config. Example:
```
variable "redwood_endpoint" { type = string default = "https://api.redwood.com" }
variable "api_key" { type = string }
provider "http" {}
resource "null_resource" "example" {
  provisioner "local-exec" {
    command = "curl -s -H 'Authorization: Bearer ${var.api_key}' ${var.redwood_endpoint}/v1/models"
  }
}
```
