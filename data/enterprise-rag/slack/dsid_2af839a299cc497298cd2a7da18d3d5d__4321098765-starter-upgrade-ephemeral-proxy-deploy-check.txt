kevin (DevEx): sample proxy snippet to include in README:

```js
// auth-proxy.js (local dev)
const http = require('http');
const fetch = require('node-fetch');
http.createServer(async (req, res) => {
  const body = await new Promise(r => { let d=''; req.on('data',c=>d+=c); req.on('end',()=>r(d)); });
  const up = await fetch('https://api.redwood.ai/v1/responses', {
    method: 'POST',
    headers: { 'Content-Type':'application/json', 'Authorization': `Bearer ${process.env.REDWOOD_EPHEMERAL_TOKEN}` },
    body
  });
  res.writeHead(up.status, { 'content-type': 'application/json' });
  res.end(await up.text());
}).listen(3000);
```
