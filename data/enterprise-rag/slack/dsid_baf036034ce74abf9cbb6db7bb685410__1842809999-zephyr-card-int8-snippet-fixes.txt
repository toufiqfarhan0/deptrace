Elliot: sample SDK snippets (please use these):
```python
from redwood import RedwoodClient
client = RedwoodClient(api_key="${RW_KEY}")
resp = client.generate(
    model="zephyr-aurora-v1.2-int8",
    input="Translate to French:\nHello world",
    options={"quantize_profile": "int8", "max_tokens": 128}
)
print(resp.text)
```
```js
import Redwood from '@redwood/inference'
const client = new Redwood({ apiKey: process.env.RW_KEY })
const r = await client.generate({ model: 'zephyr-aurora-v1.2-int8', input: '...', options: { quantize_profile: 'int8', max_tokens: 128 } })

```
```go
client := redwood.NewClient(os.Getenv("RW_KEY"))
res, err := client.Generate(ctx, redwood.GenerateRequest{
    Model: "zephyr-aurora-v1.2-int8",
    Input: "Summarize...",
    Options: map[string]interface{}{"quantize_profile": "int8", "max_tokens": 128},
})
if err != nil { log.Fatal(err) }
fmt.Println(res.Text)
```
