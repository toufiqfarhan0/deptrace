```go
// client-side pseudonymize example (pseudocode)
package examples

import (
    "crypto/hmac"
    "crypto/sha256"
    "encoding/hex"
)

func Pseudonymize(id string, key []byte) string {
    mac := hmac.New(sha256.New, key)
    mac.Write([]byte(id))
    return hex.EncodeToString(mac.Sum(nil))
}

// key should come from customer-managed KMS; do not send raw keys to Redwood.
```
