// minimal server-side broker: caches identical ephemeral exchanges for 2s
const cache = new Map();
app.get('/broker/ephemeral', async (req, res) => {
  const key = `ephemeral-${req.query.user}`;
  const hit = cache.get(key);
  const now = Date.now();
  if (hit && now - hit.ts < 2000) return res.json(hit.token);
  const token = await fetchEphemeralFromRedwood(process.env.LL_KEY);
  cache.set(key, { token, ts: now });
  setTimeout(() => cache.delete(key), 2100);
  res.json(token);
