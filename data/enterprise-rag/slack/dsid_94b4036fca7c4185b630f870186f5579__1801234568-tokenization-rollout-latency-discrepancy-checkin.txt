```sql
SELECT route, variant, count(*) AS samples, approx_percentile(latency_ms, 0.99) AS p99, avg(token_count) AS avg_tokens
FROM telemetry.events
WHERE env = 'canary' AND ts >= now() - interval '30 minutes'
GROUP BY route, variant
ORDER BY samples DESC
```
