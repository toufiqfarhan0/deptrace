// recommended backoff for 429s
async function backoff(attempt) {
  const base = 300; // ms
  const max = 5000;
  const jitter = Math.random() * 100;
  return Math.min(max, Math.floor(base * Math.pow(2, attempt)) + jitter);
}
```

mike_devx: on the Go snippet — I sketched something that uses our `Retry-After` header helper (`rl.ParseRetryAfter`) and `x-retry-count` header to preserve telemetry. can I paste?\n```go
// simplified example
import (
  "net/http"
  "time"
)

func DoWith429Handling(client *http.Client, req *http.Request) (*http.Response, error) {
  for attempt := 0; attempt < 4; attempt++ {
    resp, err := client.Do(req)
    if err != nil { return nil, err }
    if resp.StatusCode != 429 { return resp, nil }
    ra := ParseRetryAfter(resp.Header.Get("Retry-After")) // helper returns time.Duration or 0
    wait := ra
    if wait == 0 { wait = time.Duration(300*(1<<attempt))*time.Millisecond }
    time.Sleep(wait + jitter())
    // on final attempt, optionally switch to fallback model
    if attempt == 2 { // pre-switch attempt
      // mutate req to use cheaper model name
      req.Header.Set("x-redwood-fallback-model", "rwd-1.0-small")
    }
  }
  return client.Do(req)
}
```
