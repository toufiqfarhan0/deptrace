```sql
SELECT
  percentile(50, latency_ms) as p50,
  percentile(95, latency_ms) as p95,
  sample_origin
FROM requests
WHERE endpoint = '/v2/generate' AND region='eu-west'
GROUP BY sample_origin, 1h
ORDER BY time DESC
LIMIT 24;
```
