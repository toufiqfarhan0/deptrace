ivy: yes. TokenInterceptor calls refreshBlocking() which uses the same Retrofit/OkHttp client. snippet: 
```kotlin
override fun intercept(chain: Interceptor.Chain): Response {
  var token = tokenStore.get()
  if (token == null) {
    synchronized(lock) {
      if (tokenStore.get() == null) tokenStore.set(refreshBlocking())
      token = tokenStore.get()
    }
  }
  val req = chain.request().newBuilder().header("Authorization", "Bearer $token").build()
  val resp = chain.proceed(req)
  if (resp.code == 401) {
    synchronized(lock) { tokenStore.set(refreshBlocking()) }
    return chain.proceed(req.newBuilder().header("Authorization", "Bearer ${tokenStore.get()}").build())
  }
  return resp
}
```

nate: can we add a tiny Java snippet for the CompletableFuture singleflight to the samples folder? I can convert to Kotlin afterwards. gist: 
```java
private final AtomicReference<CompletableFuture<Token>> inflight = new AtomicReference<>();
public Token refreshSingleflight() {
  for (;;) {
    CompletableFuture<Token> cur = inflight.get();
    if (cur != null) return cur.join();
    CompletableFuture<Token> f = CompletableFuture.supplyAsync(() -> refreshBlocking());
    if (inflight.compareAndSet(null, f)) {
      try { return f.join(); } finally { inflight.compareAndSet(f, null); }
    }
  }
}
```
