Saira Khan: Draft module snippet: 
```
# modules/marketplace_shim/main.tf
terraform {
  required_providers {
    partner = { source = "partnercloud/partner" version = ">= 1.0, < 2.0" }
  }
}
variable "arch" { type = string default = "x86" }
variable "enable_kms" { type = bool default = false }
locals { instance_by_arch = { arm = "c7g.small" x86 = "c6i.small" } }
resource "partner_market_listing" "this" {
  name = var.name
  image = var.image
  instance_type = local.instance_by_arch[var.arch]
  region = var.region
}
resource "partner_kms_key" "optional" {
  count = var.enable_kms ? 1 : 0
  purpose = "ENCRYPTION"
}
output "listing_id" { value = partner_market_listing.this.id }
```
