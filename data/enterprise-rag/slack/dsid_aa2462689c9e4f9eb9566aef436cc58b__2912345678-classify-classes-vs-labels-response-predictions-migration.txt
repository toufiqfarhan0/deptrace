```bash
# curl canonical (labels)
curl -X POST https://api.redwood.ai/v1/classify \
  -H "Authorization: Bearer $REDWOOD_KEY" \
  -H "Content-Type: application/json" \
  -d '{"model":"classify-1.0","input":"App crashes on launch","labels":["bug","feature","question"]}'
```

```python
# python canonical (labels)
from redwood import Client
c = Client(api_key=os.environ.get("REDWOOD_KEY"))
resp = c.classify(model="classify-1.0", input="App crashes on launch", labels=["bug","feature","question"])
# resp.predictions -> [{"label":..., "confidence":...}]
```

```javascript
// js migration snippet (accepts classes -> maps to labels)
import Redwood from '@redwood/inference'
const client = new Redwood({ apiKey: process.env.REDWOOD_KEY })
// legacy callers might still pass `classes`
await client.classify({ model: 'classify-1.0', input: 'App crashes on launch', classes: ['bug','feature'] })
// SDK will map `classes` -> `labels` and log a deprecation warning in dev builds
```
