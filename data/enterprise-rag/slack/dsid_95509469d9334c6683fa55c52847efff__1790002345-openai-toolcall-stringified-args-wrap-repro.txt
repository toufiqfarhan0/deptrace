jon-eng: For reference, quick JS tolerant parse snippet I used during repro:
```js
function tolerantParseToolCall(tc) {
  let args = tc.arguments;
  if (typeof args === 'string') {
    try { args = JSON.parse(args); } catch (e) { return { error: 'arguments not parseable' }; }
  }
  // normalize enums -> lowercase keys/values as needed
  if (args.currency && typeof args.currency === 'string') args.currency = args.currency.toLowerCase();
  if (args.amount && typeof args.amount === 'string') args.amount = parseFloat(args.amount);
  return { name: tc.name, arguments: args };
}
```
