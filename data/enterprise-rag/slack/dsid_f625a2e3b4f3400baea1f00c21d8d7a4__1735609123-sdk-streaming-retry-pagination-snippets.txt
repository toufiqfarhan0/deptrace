```ts
// TS: event-streaming + AbortController + retry helper
import { RedwoodClient } from '@redwood/inference';
const client = new RedwoodClient({ apiKey: process.env.RW_KEY });

async function streamPrompt(prompt: string) {
  const ac = new AbortController();
  const timer = setTimeout(() => ac.abort(), 45_000);
  try {
    const it = client.responses.stream({ prompt }, { signal: ac.signal });
    for await (const ev of it) {
      if (ev.type === 'delta') process.stdout.write(ev.delta);
      if (ev.type === 'meta') console.error('[meta]', ev.info);
      if (ev.type === 'end') console.log('\n-- stream ended --');
    }
  } finally {
    clearTimeout(timer);
  }
}

async function withRetries(fn: () => Promise<any>, attempts = 3) {
  let last;
  for (let i = 0; i < attempts; i++) {
    try { return await fn(); } catch (e) { last = e; await new Promise(r => setTimeout(r, Math.pow(2, i) * 100)); }
  }
  throw last;
}
```

lee: Python draft:
```python
# Python: streaming with backoff and compatibility with old/new event shapes
import time
from redwood import Client
client = Client(api_key="${RW_KEY}")

def stream_with_retries(prompt, max_tries=4):
    for attempt in range(1, max_tries + 1):
        try:
            for ev in client.stream_generate(prompt):
                # ev may be a simple string, or an object with .delta / .text
                txt = getattr(ev, 'delta', None) or getattr(ev, 'text', None) or (ev if isinstance(ev, str) else '')
                print(txt, end='', flush=True)
            return
        except (ConnectionError, TimeoutError):
            time.sleep(min(2 ** attempt, 30))
    raise RuntimeError('stream failed after retries')
```
