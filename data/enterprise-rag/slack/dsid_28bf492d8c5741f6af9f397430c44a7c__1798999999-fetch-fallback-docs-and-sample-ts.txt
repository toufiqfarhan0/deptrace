oliver: Proposed tiny runtime helper (in thread):
```ts
// runtime/fetch-or-undici.ts
let cached: typeof fetch | undefined;
export async function runtimeFetch(): Promise<typeof fetch> {
  if (cached) return cached;
  if ((globalThis as any).fetch) return (cached = globalThis.fetch);
  try {
    const mod = await import('undici');
    return (cached = (mod as any).fetch);
  } catch (_) {
    try {
      // CJS fallback
      // eslint-disable-next-line @typescript-eslint/no-var-requires
      const req = require('undici');
      return (cached = req.fetch);
    } catch (e) {
      throw new Error('No fetch available. Install undici or run on Node 18+.');
    }
  }
}
```
