# recommended pattern for transient 429s/timeouts
import asyncio
import random

async def retry_with_backoff(fn, max_attempts=5, base=0.5, cap=5.0):
    attempt = 0
    elapsed = 0.0
    while attempt < max_attempts:
        try:
            return await fn()
        except TransientError:
            attempt += 1
            wait = min(cap, base * (2 ** (attempt - 1)))
            jitter = random.uniform(0, wait * 0.3)
            sleep = wait + jitter
            elapsed += sleep
            if elapsed > 10.0:
                raise
            await asyncio.sleep(sleep)
```
